uniform 	vec4 hlslcc_mtx4x4unity_ObjectToWorld[4];
uniform 	vec4 hlslcc_mtx4x4unity_MatrixVP[4];
in highp vec4 in_POSITION0;
in highp vec2 in_TEXCOORD0;
out highp vec2 vs_TEXCOORD0;
out highp vec3 vs_TEXCOORD1;
vec4 u_xlat0;
vec4 u_xlat1;
void main()
{
    vs_TEXCOORD0.xy = in_TEXCOORD0.xy;
    u_xlat0 = in_POSITION0.yyyy * hlslcc_mtx4x4unity_ObjectToWorld[1];
    u_xlat0 = hlslcc_mtx4x4unity_ObjectToWorld[0] * in_POSITION0.xxxx + u_xlat0;
    u_xlat0 = hlslcc_mtx4x4unity_ObjectToWorld[2] * in_POSITION0.zzzz + u_xlat0;
    vs_TEXCOORD1.xyz = hlslcc_mtx4x4unity_ObjectToWorld[3].xyz * in_POSITION0.www + u_xlat0.xyz;
    u_xlat0 = u_xlat0 + hlslcc_mtx4x4unity_ObjectToWorld[3];
    u_xlat1 = u_xlat0.yyyy * hlslcc_mtx4x4unity_MatrixVP[1];
    u_xlat1 = hlslcc_mtx4x4unity_MatrixVP[0] * u_xlat0.xxxx + u_xlat1;
    u_xlat1 = hlslcc_mtx4x4unity_MatrixVP[2] * u_xlat0.zzzz + u_xlat1;
    gl_Position = hlslcc_mtx4x4unity_MatrixVP[3] * u_xlat0.wwww + u_xlat1;
    return;
}

#ifdef FRAGMENT
#version 300 es

precision highp float;
precision highp int;
#define HLSLCC_ENABLE_UNIFORM_BUFFERS 1
#if HLSLCC_ENABLE_UNIFORM_BUFFERS
#define UNITY_UNIFORM
#else
#define UNITY_UNIFORM uniform
#endif
#define UNITY_SUPPORTS_UNIFORM_LOCATION 1
#if UNITY_SUPPORTS_UNIFORM_LOCATION
#define UNITY_LOCATION(x) layout(location = x)
#define UNITY_BINDING(x) layout(binding = x, std140)
#else
#define UNITY_LOCATION(x)
#define UNITY_BINDING(x) layout(std140)
#endif
uniform 	vec4 _Time;
uniform 	vec4 _NoiseTex_ST;
uniform 	vec4 _DistortTex_ST;
uniform 	float _NoiseCutOff;
uniform 	float _NoiseSmooth;
uniform 	float _FlowSpeed;
uniform 	float _DistortSpeed;
uniform 	float _DistortAmount;
uniform 	mediump vec4 _Color;
uniform 	mediump vec4 _Color2;
uniform 	mediump vec4 _NoiseColor;
uniform 	float _TopFade;
uniform 	float _BottomFade;
uniform 	vec2 _WorldScale;
UNITY_LOCATION(0) uniform mediump sampler2D _DistortTex;
UNITY_LOCATION(1) uniform mediump sampler2D _NoiseTex;
UNITY_LOCATION(2) uniform mediump sampler2D _MainTex;
in highp vec2 vs_TEXCOORD0;
in highp vec3 vs_TEXCOORD1;
layout(location = 0) out mediump vec4 SV_Target0;
vec4 u_xlat0;
mediump vec4 u_xlat16_0;
vec2 u_xlat1;
mediump vec4 u_xlat16_1;
mediump float u_xlat16_2;
float u_xlat3;
vec2 u_xlat4;
float u_xlat5;
mediump vec3 u_xlat16_6;
float u_xlat8;
mediump float u_xlat16_8;
vec2 u_xlat9;
void main()
{
    u_xlat0.xy = _Time.yy * vec2(_DistortSpeed, _FlowSpeed);
    u_xlat0.xy = (-u_xlat0.xy);
    u_xlat1.x = vs_TEXCOORD1.x * _WorldScale.xxxy.z;
    u_xlat1.y = vs_TEXCOORD1.y * (-_WorldScale.xxxy.w);
    u_xlat9.xy = u_xlat1.xy * _DistortTex_ST.xy + _DistortTex_ST.zw;
    u_xlat1.xy = u_xlat1.xy * _NoiseTex_ST.xy + _NoiseTex_ST.zw;
    u_xlat0.z = 0.0;
    u_xlat0.xw = u_xlat0.zx + u_xlat9.xy;
    u_xlat4.xy = u_xlat0.zy + u_xlat1.xy;
    u_xlat16_0.xw = texture(_DistortTex, u_xlat0.xw).xy;
    u_xlat0.xw = u_xlat16_0.xw * vec2(2.0, 2.0) + vec2(-1.0, -1.0);
    u_xlat0.xy = u_xlat0.xw * vec2(_DistortAmount) + u_xlat4.xy;
    u_xlat16_8 = texture(_NoiseTex, u_xlat0.xy).x;
    u_xlat16_1 = texture(_MainTex, u_xlat0.xy);
    u_xlat0.x = (-_NoiseSmooth) + _NoiseCutOff;
    u_xlat4.x = (-u_xlat0.x) + u_xlat16_8;
    u_xlat8 = _NoiseSmooth + _NoiseCutOff;
    u_xlat0.x = (-u_xlat0.x) + u_xlat8;
    u_xlat0.x = float(1.0) / u_xlat0.x;
    u_xlat0.x = u_xlat0.x * u_xlat4.x;
    u_xlat0.x = clamp(u_xlat0.x, 0.0, 1.0);
    u_xlat4.x = u_xlat0.x * -2.0 + 3.0;
    u_xlat0.x = u_xlat0.x * u_xlat0.x;
    u_xlat0.x = u_xlat0.x * u_xlat4.x;
    u_xlat16_2 = u_xlat0.x * _NoiseColor.w;
    u_xlat0 = (-_Color) + _Color2;
    u_xlat3 = (-vs_TEXCOORD0.y) + 1.0;
    u_xlat0 = vec4(u_xlat3) * u_xlat0 + _Color;
    u_xlat16_6.xyz = (-u_xlat0.xyz) * u_xlat16_1.xyz + _NoiseColor.xyz;
    u_xlat16_0 = u_xlat16_1 * u_xlat0;
    SV_Target0.xyz = vec3(u_xlat16_2) * u_xlat16_6.xyz + u_xlat16_0.xyz;
    u_xlat1.xy = vec2(1.0, 1.0) / vec2(_TopFade, _BottomFade);
    u_xlat5 = u_xlat1.y * u_xlat3;
    u_xlat5 = clamp(u_xlat5, 0.0, 1.0);
    u_xlat1.x = u_xlat1.x * vs_TEXCOORD0.y;
    u_xlat1.x = clamp(u_xlat1.x, 0.0, 1.0);
    u_xlat9.x = u_xlat5 * -2.0 + 3.0;
    u_xlat5 = u_xlat5 * u_xlat5;
    u_xlat5 = u_xlat5 * u_xlat9.x;
    u_xlat9.x = u_xlat1.x * -2.0 + 3.0;
    u_xlat1.x = u_xlat1.x * u_xlat1.x;
    u_xlat1.x = u_xlat1.x * u_xlat9.x;
    u_xlat1.x = u_xlat5 * u_xlat1.x;
    u_xlat1.x = u_xlat16_0.w * u_xlat1.x;
    SV_Target0.w = u_xlat1.x;
    return;
}
